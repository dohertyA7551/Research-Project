merged_df_scores <- merged_df %>%
  mutate(
    # summed abundance over bad (event) metaclusters
    bad_abund  = rowSums(across(all_of(mc_event_both)),    na.rm = TRUE),
    # summed abundance over good (no-event) metaclusters
    good_abund = rowSums(across(all_of(mc_no_event_both)), na.rm = TRUE)
  ) %>%
  mutate(
    # raw score: higher = more bad vs good
    score_raw = bad_abund - good_abund
  )

mc_prob_stats_adjuvent <- map_dfr(leiden_cols, function(col) {
  p <- merged_df_dfs[[col]]

  p_event   <- p[merged_df_dfs$dfs_group_adjuv == "DFS event"]
  p_noevent <- p[merged_df_dfs$dfs_group_adjuv == "No DFS event"]

  p_event   <- p_event[!is.na(p_event)]
  p_noevent <- p_noevent[!is.na(p_noevent)]

  if (length(p_event) == 0 || length(p_noevent) == 0) {
    return(tibble(
      metacluster          = col,
      mean_prob_event      = NA_real_,
      mean_prob_noevent    = NA_real_,
      median_prob_event    = NA_real_,
      median_prob_noevent  = NA_real_,
      p_value              = NA_real_
    ))
  }

  wt <- wilcox.test(p_event, p_noevent, exact = FALSE)

  tibble(
    metacluster          = col,
    mean_prob_event      = mean(p_event),
    mean_prob_noevent    = mean(p_noevent),
    median_prob_event    = median(p_event),
    median_prob_noevent  = median(p_noevent),
    p_value              = wt$p.value
  )
})
